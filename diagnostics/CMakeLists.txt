cmake_minimum_required(VERSION 3.21)
project(inputalign_diagnostics VERSION 0.3.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
option(INPUTALIGN_BUILD_MOD "Build Windows Geode companion" ON)
include(CTest)
if(BUILD_TESTING)
    add_executable(diagnostics_test tests/diagnostics_test.cpp)
    target_include_directories(diagnostics_test PRIVATE src)
    add_test(NAME diagnostics_core COMMAND diagnostics_test)
    add_executable(landing_tap_test tests/landing_tap_test.cpp)
    target_include_directories(landing_tap_test PRIVATE src)
    add_test(NAME landing_tap COMMAND landing_tap_test)
    add_executable(tap_eligibility_test tests/tap_eligibility_test.cpp)
    target_include_directories(tap_eligibility_test PRIVATE src)
    add_test(NAME tap_eligibility COMMAND tap_eligibility_test)
endif()
if(INPUTALIGN_BUILD_MOD)
    if(NOT DEFINED ENV{GEODE_SDK})
        message(FATAL_ERROR "GEODE_SDK must point to Geode SDK v5.11.2")
    endif()
    add_library(${PROJECT_NAME} SHARED src/main.cpp)
    add_subdirectory($ENV{GEODE_SDK} ${CMAKE_CURRENT_BINARY_DIR}/geode)
    setup_geode_mod(${PROJECT_NAME})
endif()
